# launch a qlogin session on sge

# 1: set the singularity path variable. for pointing to the singularity container

SINGULARITY_PATH=/mnt/research/tools/LINUX/MITOHPC/MitoHPC

# 2: launch the mitohpc in interactive shell mode

singularity shell -B /mnt:/mnt --env HP_SDIR="/MitoHPC/scripts/" $SINGULARITY_PATH/mitohpc.simg --pwd 

# 3: 

cd to directory where the subdirectory containing the CRAM files are located at.

# 4: copy the init.sh from the container to the working directory (what you just cd'd into)

copy /MitoHPC/scripts/init.sh to current working directory

# 5:

change input directory in the init.sh script to wherever you have the cram files located at (change $PWD in line 121 in init.sh)

export HP_ADIR=$PWD/CRAM/	# bams or crams input file directory

technically i believe you should be able to leave it as is if following the directions above, but in case you are in a differet directory for whatever reason.

# 6: launch init.sh as below which will create some new files.

. ./init.sh

# 7: run the below to create the execution script

$HP_SDIR/run.sh > run.all.sh
